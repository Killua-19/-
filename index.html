const canvas = document.getElementById("matrix"); const ctx = canvas.getContext("2d"); canvas.width = window.innerWidth; canvas.height = window.innerHeight; const emoji = "🪼🪼🪼🪼"; const fontSize = 24; let columns = Math.floor(canvas.width / (fontSize * 4.5)); let drops = Array.from({ length: columns }, () => Math.random() * -40); const explosiones = []; function draw() { ctx.fillStyle = "rgba(0,0,0,0.25)"; ctx.fillRect(0,0,canvas.width,canvas.height); ctx.font = fontSize + "px Arial Black"; ctx.shadowColor = "#ffffff"; ctx.shadowBlur = 14; for(let i = 0; i < drops.length; i++){ ctx.fillStyle = "#ffffff"; ctx.fillText( emoji, i * fontSize * 4.5, drops[i] * fontSize ); for(let j = 1; j < 6; j++){ ctx.fillStyle = `rgba(255,255,255,${1-j*0.15})`; ctx.fillText( emoji, i * fontSize * 4.5, (drops[i]-j) * fontSize ); } drops[i] += 0.45; if(drops[i]*fontSize > canvas.height){ drops[i] = Math.random()*-35; } } for(let idx = explosiones.length-1; idx>=0; idx--){ const exp = explosiones[idx]; ctx.font = "22px Arial Black"; ctx.shadowBlur = 16; ctx.fillStyle = `rgba(255,255,255,${exp.life/50})`; ctx.fillText("🪼", exp.x, exp.y); exp.x += exp.dx; exp.y += exp.dy; exp.life--; if(exp.life<=0){ explosiones.splice(idx,1); } } requestAnimationFrame(draw); } function explosion(x,y){ for(let i=0;i<16;i++){ const angle=Math.random()*Math.PI*2; const velocity=2+Math.random()*6; explosiones.push({ x:x, y:y, dx:Math.cos(angle)*velocity, dy:Math.sin(angle)*velocity, life:55 }); } } document.addEventListener("click",e=>{ explosion(e.clientX,e.clientY); }); document.addEventListener("touchstart",e=>{ const t=e.touches[0]; explosion(t.clientX,t.clientY); }); window.addEventListener("resize",()=>{ canvas.width=window.innerWidth; canvas.height=window.innerHeight; columns=Math.floor(canvas.width/(fontSize*4.5)); drops=Array.from( {length:columns}, ()=>Math.random()*-40 ); }); requestAnimationFrame(draw);
